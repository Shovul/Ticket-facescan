Cấu hình sự kiện: ${item.eventTitle}
${item.feePercent.toFixed(1)}%
0.0% 8.0%
Nhập 0% nếu là sự kiện nội bộ sinh viên trường.
${currentAdvanceRate}%
10% 60%
`; openModal('configModal'); } // Hàm ẩn/hiện slider tùy theo giá trị select function toggleSettleModel(val) { const wrapper = document.getElementById('advanceSliderWrapper'); if (val === 'multi') { wrapper.style.display = 'flex'; } else { wrapper.style.display = 'none'; } } function saveEventConfig() { const item = financeData.find(i => i.id === targetTxnId); if (!item) return; const newFee = parseFloat(document.getElementById('cfgFeePercent').value); const newModel = document.getElementById('cfgSettleModel').value; if (isNaN(newFee) || newFee < 0) { alert("Vui lòng nhập tỷ lệ phí hợp lệ!"); return; } item.feePercent = newFee; item.settleModel = newModel; // Nếu chọn multi thì lưu giá trị slider vào item if (newModel === 'multi') { const advRate = parseInt(document.getElementById('cfgAdvanceRate').value, 10); item.advanceRate = advRate; } else { item.advanceRate = 0; } closeModal('configModal'); renderFinanceTable(); showToast(`Đã lưu cấu hình tài chính mới thành công!`); }
